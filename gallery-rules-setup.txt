match /cheonsal_gallery/{photoId} {
  allow read: if true;
  allow create: if request.auth != null
    && request.auth.uid == '4wsmGJicudaNhZh0329QQVDZy112'
    && request.resource.data.keys().hasOnly(['title','category','description','url','path','createdAt','uid'])
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.title is string && request.resource.data.title.size() > 0 && request.resource.data.title.size() <= 80
    && request.resource.data.description is string && request.resource.data.description.size() <= 300
    && request.resource.data.category in ['단체사진','레이드·보스','점령전·제전','득템 인증','웃긴 순간','기타']
    && request.resource.data.url is string && request.resource.data.url.size() < 2048
    && request.resource.data.path is string && request.resource.data.path.matches('cheonsal_gallery/' + request.auth.uid + '/[^/]+')
    && request.resource.data.createdAt == request.time;
  allow delete: if request.auth != null && request.auth.uid == '4wsmGJicudaNhZh0329QQVDZy112';
  allow update: if false;
}
